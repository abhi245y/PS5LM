# PS5LM's build: llama.cpp as a subproject, plus the ps5lm payload.
#
# scripts/build-llama.sh configures this with the ps5-payload-dev SDK's
# toolchain. llama.cpp's own targets (llama-server, llama-cli, ...) are built
# from here too, so there is one build tree.

cmake_minimum_required(VERSION 3.20)
project(ps5lm C CXX)

set(LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/llama.cpp")
# All programs in build/.../bin, as when llama.cpp is built on its own.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory(${LLAMA_DIR} llama.cpp)

# The tools leave through _Exit when main returns, skipping Sony's exit() and
# the static destructors it runs (compat/exit.cpp, issue #3).
foreach(tool llama-cli llama-server llama-bench llama-completion)
    if(TARGET ${tool})
        target_sources(${tool} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/compat/exit.cpp")
        target_link_options(${tool} PRIVATE "LINKER:--wrap=main")
    endif()
endforeach()

# Files the payload carries inside itself (the pages and the CA bundle), as
# byte arrays in a generated header. Reconfigures when one of them changes.
function(ps5lm_embed out_header)
    set(content "// Generated by ps5/CMakeLists.txt, do not edit.\n#pragma once\n#include <cstddef>\n")
    foreach(pair ${ARGN})
        string(REPLACE "=" ";" kv "${pair}")
        list(GET kv 0 name)
        list(GET kv 1 path)
        file(READ "${path}" hex HEX)
        string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
        file(SIZE "${path}" size)
        string(APPEND content
            "static const unsigned char ${name}[] = {${bytes}0x00};\n"
            "static const std::size_t ${name}_len = ${size};\n")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    endforeach()
    file(WRITE "${out_header}" "${content}")
endfunction()

ps5lm_embed("${CMAKE_CURRENT_BINARY_DIR}/ps5lm_assets.h"
    "chat_html=${CMAKE_CURRENT_SOURCE_DIR}/webui/index.html"
    "library_html=${CMAKE_CURRENT_SOURCE_DIR}/ps5lm/library.html"
    "ca_bundle=${CMAKE_CURRENT_SOURCE_DIR}/ps5lm/cacert.pem")

add_executable(ps5lm ps5lm/app.cpp)
target_include_directories(ps5lm PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${LLAMA_DIR}/vendor")
# Sony's system and user services, for opening the browser.
target_link_libraries(ps5lm PRIVATE llama-server-impl cpp-httplib SceSystemService SceUserService)
target_compile_features(ps5lm PRIVATE cxx_std_17)
set_target_properties(ps5lm PROPERTIES OUTPUT_NAME ps5lm SUFFIX ".elf")
